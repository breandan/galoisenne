    \mysection{Main Idea}

    \vspace*{-0.8cm}
    \null\hspace*{2.8cm}\begin{minipage}[c]{0.86\columnwidth}
          \renewcommand\labelitemi{$\vcenter{\hbox{\small\bullet}}$}
          \begin{itemize}
            \labelsep=0.8cm
%            \itemindent=-0.6cm
            \item \textbf{Assume}: we are given a learned distribution, $P_{\theta}(\sigma)$, with support over $\sigma: \Sigma^+$ but largely concentrated within $\mathcal{L}(G)$, where $G$ is a known CFG.
            \item \textbf{Our goal}: distill a surrogate WFA, $\alpha_{\hat\theta}$, such that for all $\sigma \in \mathcal{L}(G)$, $P_{\hat\theta}(\sigma) \propto P_{\theta}\big(\sigma \mid \sigma \in \mathcal{L}(G)\big)$, while preserving inclusion $\big(\mathcal{L}(\alpha)\supseteq \mathcal{L}(G)\big)$.
            \item \textbf{Desiderata}: The regular description $\alpha$ should be reasonably concise and precise, i.e., $\Delta = |\mathcal{L}(\alpha) \setminus \mathcal{L}(G)|$ is approximately minimized up to $\Sigma^n$.
          \end{itemize}
    \end{minipage}

\mysection{Nederhof's Construction}

\null\hspace*{2.8cm}\begin{minipage}[c]{0.87\columnwidth}
  A recursive transition network (RTN) of a CFG $\langle \Sigma,V, P, S\rangle$ is an automaton,
  \[
    \mathcal R=\big\{R_A=(Q_A,\delta_A,\alpha_A,F_A)\big\}_{A\in V},
    \:\:
    \delta_A\subseteq Q_A\times\big(\Sigma\cup V\cup\{\varepsilon\}\big)\times Q_A,
  \]
  where an arc labelled \(B\in V\) denotes a call to \(R_B\). For example, consider the CFG, $\texttt{S } \rightarrow \texttt{ x } \mid \texttt{ S S } \mid \texttt{ \{ S \}}$, which has the following RTN:\vspace{0.6cm}

  \textbf{Nederhof's \(d\)-RTN approximation} is a depth-bounded RTN unfolding. A nonterminal arc \(p \overset{B}{\longrightarrow} q\) is replaced by a fresh copy of the RTN for \(B\), adding \(\varepsilon\)-arcs from \(p\) to the copy's initial state and the copy's final state back to \(q\): \( p \overset{\varepsilon}{\longrightarrow} \alpha_B', \:\: \omega_B' \overset{\varepsilon}{\longrightarrow} q \quad(\forall \omega_B'\in F_B')\). Unfold calls for \(d\) rounds. At depth \(d\), every remaining nonterminal arc \(p\overset{B}{\longrightarrow}q\) is woven into a shared copy of \(R_B\): \( p \overset{\varepsilon}{\longrightarrow} \alpha_B, \:\: \omega_B \overset{\varepsilon}{\longrightarrow} q \quad(\forall \omega_B\in F_B) \).     \null\hspace*{2.8cm}\begin{minipage}[c]{0.86\columnwidth}
  We can measure over-approximation tightness by considering finite models,\\
    \[
      \rho(\alpha, G, m) =
      \frac{\sum_{n=1}^m |\mathcal{L}(G)\cap\Sigma^n|}
      {\sum_{n=1}^m |\mathcal{L}(\alpha)\cap\Sigma^n|},
      \:\:\alpha \sqsubset_G^m \alpha'
      \iff
      \rho(\alpha',G,m) < \rho(\alpha,G,m).
    \]
  \end{minipage}
    \vspace{0.7em}

    \mysection{Method}

    \null\hspace*{2.8cm}\begin{minipage}[c]{0.86\columnwidth}
    We propose a counterexample-guided abstraction refinement (CEGAR) loop that incrementally refines $\alpha_{\hat\theta}$ until a desired precision or $|Q|_\max$ is reached:\vspace{0.4cm}
    \end{minipage}

    \begin{enumerate}
      \itemsep0em
      \itemindent=4cm
      \sloppy
      \setlength{\emergencystretch}{5cm}
      \item \phantom{.}Initialize $\alpha_{\hat\theta}'\leftarrow \alpha_d$ using Nederhof's $d$-RTN construction.
      \item \phantom{.}Propose a refinement $D_\text{KL}(P_{\theta}\|\alpha_{\hat\theta}) < D_\text{KL}(P_{\theta}\|\alpha_{\hat\theta}')$ and $\alpha_{\hat\theta} \sqsubset_G^m \alpha_{\hat\theta}'$.
      \item \phantom{.}Binarize $\alpha_\hat\theta$ and verify $\mathcal{L}(\alpha) \supset \mathcal{L}(G)$ by checking $\varnothing =\mathcal{L}(G)\cap \overline{\mathcal{L}(\alpha)}$.
      \item \phantom{.}If verification fails, collect $x \in \mathcal{L}(G)\cap \overline{\mathcal{L}(\alpha)}$, otherwise go to step 2.
      \item \phantom{.}Repair $\alpha$ restoring inclusion, renormalize as $\alpha_{\hat\theta}'$, then go to step 2.
    \end{enumerate}

    \mysection{Intersection Decoding}

    \vspace{-2.5cm}
    \null\hspace*{2.8cm}\begin{minipage}[t]{0.86\columnwidth}

%      across the validation set. Optionally, we may rerank the top-1,000 repairs using a Levenshtein-aligned transformer reranker (LATR) which takes a word and a list of candidates and estimates, $P_\theta(\sigma \mid \err \sigma)$.\\
      \begin{definition}[Bounded Levenshtein-CFL reachability]\label{def:bcflr}
        Given a CFL, $\ell$, and an invalid string, $\err{\sigma}: \bar\ell$, find every valid string reachable within $d$ edits of $\err{\sigma}$, i.e., if $\Delta_L$ is the Levenshtein metric and $\mathcal{L}\big(\text{L}(\err\sigma, d)\big) = \{\sigma' \mid \Delta_L(\err{\sigma}, \sigma') \leq d\}$ is the Levenshtein $d$-ball, we seek to find $\ell_\cap = \mathcal{L}\big(\text{L}(\err\sigma, d)\big) \cap \ell$.
      \end{definition}

      \vspace{0.1cm}Moreover, we want a procedure that surfaces \textit{both} natural and valid repairs:\vspace{0.1cm}

      \begin{definition}[Ranked repair]\label{def:ranked-repair}
        Given a finite language $\ell_\cap = \mathcal{L}\big(L(\err\sigma, d)\big) \cap \ell$ and LM, $\text{P}_\theta: \Sigma^* \rightarrow \mathbb{R}$, find the top-$k$ maximum probability repairs, i.e.,
        \begin{equation*}
          R(\ell_\cap, P_\theta): 2^{\Sigma^+} \times (\Sigma^+ \rightarrow \mathbb{R}) \rightarrow (\Sigma^+)^{\leq k} = \argmax_{\bm{\sigma} \subseteq \ell_\cap, |\bm{\sigma}| \leq k} \sum_{\sigma \in \bm{\sigma}}\text{P}_\theta(\sigma)
        \end{equation*}
      \end{definition}
      \end{minipage}

    \mysection{Evaluation}
    \null\hspace*{2.8cm}\begin{minipage}[c]{0.86\columnwidth}
    We will compare four different methods for constrained decoding from $\ell_\cap$:

    \begin{enumerate}
      \itemsep0em
      \itemindent=4cm
      \sloppy
      \setlength{\emergencystretch}{5cm}
      \item \phantom{.} A \textbf{4-gram} model with Laplace smoothing trained on $\sigma \in \mathcal{L}(G)$
      \item \phantom{.} A \textbf{GPT}-style model trained on $\err\sigma\sigma'$ with LTR $\sigma'$ masking
      \item \phantom{.} A \textbf{PDFA} distilled from $P_\theta$ overapproximating $\mathcal{L}(G)$
      \item \phantom{.} A \textbf{LATeR} (PDFA decoding + top-512 reranker)
    \end{enumerate}
      Let $\text{E}_\theta: \Sigma^+ \rightarrow \mathbb{R}^p$ be a transformer encoder and $E'_\theta: \Sigma^+ \times \mathbb{Z}_4^+ \rightarrow \mathbb{R}^p$ be an encoder which takes in a Levenshtein alignment $(\text{LA}: \Sigma^+ \times \Sigma^+ \rightarrow \mathbb{Z}_4^+)$ for each $\langle\err\sigma, \sigma\rangle$ pair. Now add an MLP and output a scalar relevance score:
      \begin{equation*}
        \Delta_\theta(\err\sigma, \sigma) = \text{MLP}_{\theta}\Big(\text{E}_\theta\big(\err\sigma\big), \text{E}_\theta'\big(\sigma, \text{LA}(\err\sigma, \sigma)\big)\Big)
      \end{equation*}

      We call this model a Levenshtein-Aligned Transformer Reranker (LATeR) and train on $\langle\err\sigma, \sigma'\rangle$ with PDFA-sampled confounders and a listwise CE loss.
    \end{minipage}

    \mysection{Conclusion}

    \null\hspace*{2.8cm}\begin{minipage}[c]{0.86\columnwidth}
        \renewcommand\labelitemi{$\vcenter{\hbox{\small\bullet}}$}
        \begin{itemize}[labelsep=0.8cm]
          \item Single-threaded PDFA sampling exhibits $10^3-10^5\times$ higher throughput than constrained LLM decoding, often saturating $\ell_\cap$ in under 100ms.
          \item The size-k subset maximizing $P_{\theta}$ over $\ell_\cap$ can far outweigh the mass from an LLM-guided subsampling. We call this the \textbf{decoder-retriever gap}.
          \item Hence, exhaustive ranking with a low-fidelity but fast $P_\theta$ approximation is preferable to decoding a highly calibrated but low-throughput LLM.
          \item Is there a succinct construction (possibly with a very different topology) that soundly overapproximates $\mathcal{L}(G)$ and more faithfully reconstructs $P_\theta$?
        \end{itemize}
    \end{minipage}
